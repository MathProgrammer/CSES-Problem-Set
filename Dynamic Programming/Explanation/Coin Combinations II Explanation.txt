It is same like Coin Combinations I, but coin values are different. 

f(i) = sum(f(i - c))

-----

int main()
{
    int no_of_coins, total;
    cin >> no_of_coins >> total;

    vector <int> value(no_of_coins + 1);
    for(int i = 1; i <= no_of_coins; i++)
    {
        cin >> value[i];
    }

    const int MOD = 1e9 + 7;
    vector <long long> no_of_ways(total + 1);
    no_of_ways[0] = 1;

    for(int c = 1; c <= no_of_coins; c++)
    {
        for(int i = 0; i <= total; i++)
        {
            if(value[c] <= i)
            {
                no_of_ways[i] += no_of_ways[i - value[c]];
            }

            no_of_ways[i] %= MOD;
        }
    }

    cout << no_of_ways[total] << "\n";

    return 0;
}
